/**
 * Model revision tracking for frees.
 *
 * Correlates Check, Solve, and Table requests with monotonic document/model revisions.
 * Any user edit (text change, table update, slider/REPL override change, complex mode toggle,
 * or project load) increments the revision.
 * Responses returned for obsolete revisions are discarded so that in-flight solves do not
 * overwrite newer state or display stale results.
 */

export interface ModelRevisionState {
  currentRevision: number
  lastCheckedRevision: number | null
  lastCheckSolvable: boolean
  checkInFlight: boolean
  checkInvalidatedWhileInFlight: boolean
}

export class ModelRevisionTracker {
  private rev = 1
  private lastCheckedRev: number | null = null
  private lastCheckSolv = false
  private inFlight = false
  private invalidatedWhileInFlight = false

  /** The current monotonic model revision. */
  get current(): number {
    return this.rev
  }

  /** Advances the model revision when inputs change (text, tables, overrides, complex mode, project load). */
  bump(): number {
    this.rev += 1
    if (this.inFlight) {
      this.invalidatedWhileInFlight = true
    }
    return this.rev
  }

  /** Checks if a given revision is still the current active model revision. */
  isCurrent(revision: number): boolean {
    return revision === this.rev
  }

  /** Marks that a check request for a specific revision has started. */
  startCheck(): number {
    this.inFlight = true
    return this.rev
  }

  /**
   * Records the completion of a check for a specific revision.
   * Returns true if this check result belongs to the current revision and was accepted.
   */
  finishCheck(revision: number, solvable: boolean): { isCurrent: boolean; needsRecheck: boolean } {
    this.inFlight = false
    const accepted = revision === this.rev
    if (accepted) {
      this.lastCheckedRev = revision
      this.lastCheckSolv = solvable
    }
    const needsRecheck = this.invalidatedWhileInFlight
    this.invalidatedWhileInFlight = false
    return { isCurrent: accepted, needsRecheck }
  }

  /** Invalidates any previously cached check result without bumping revision. */
  invalidateCheck(): void {
    this.lastCheckedRev = null
    this.lastCheckSolv = false
    if (this.inFlight) {
      this.invalidatedWhileInFlight = true
    }
  }

  /**
   * Returns true if the system has a valid check result that passes for the CURRENT revision.
   */
  isCheckValidAndSolvable(): boolean {
    return this.lastCheckedRev === this.rev && this.lastCheckSolv
  }

  /**
   * Gets the last checked revision.
   */
  get lastCheckedRevision(): number | null {
    return this.lastCheckedRev
  }

  /** Snapshot of current state for debugging/inspection. */
  snapshot(): ModelRevisionState {
    return {
      currentRevision: this.rev,
      lastCheckedRevision: this.lastCheckedRev,
      lastCheckSolvable: this.lastCheckSolv,
      checkInFlight: this.inFlight,
      checkInvalidatedWhileInFlight: this.invalidatedWhileInFlight,
    }
  }
}
